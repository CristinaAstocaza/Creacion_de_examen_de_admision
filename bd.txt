Create table categoria_examen(
id_categoria serial primary key,
nombre varchar(200) not null,
descripcion varchar(250),
estado varchar(100) not null DEFAULT 'ACTIVO',
fecha_creacion timestamp DEFAULT current_timestamp,
fecha_actualizacion timestamp
);

CREATE TABLE curso (
    id_curso SERIAL PRIMARY KEY,
    id_categoria INT NOT NULL,
    nombre VARCHAR(200) NOT NULL,
    descripcion VARCHAR(120),
    estado VARCHAR(100) NOT NULL DEFAULT 'ACTIVO',
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_curso_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria_examen(id_categoria),
    CONSTRAINT uk_categoria_curso
        UNIQUE (id_categoria, nombre)
);

CREATE TABLE pregunta (
	id_pregunta SERIAL PRIMARY KEY,
	id_curso INT NOT NULL,
	codigo VARCHAR(20) NOT NULL UNIQUE,
	enunciado TEXT NOT NULL,
	imagen_enunciado_url TEXT,
	dificultad VARCHAR(30) NOT NULL,
	estado VARCHAR(20) NOT NULL DEFAULT 'ACTIVA',
	fecha_creacion timestamp DEFAULT current_timestamp,
	fecha_actualizacion timestamp,

	CONSTRAINT fk_pregunta_curso FOREIGN KEY (id_curso) 
	REFERENCES curso(id_curso),

	CONSTRAINT chk_dificultad CHECK (dificultad IN ('FACIL','MEDIO','DIFICIL')),

	CONSTRAINT chk_estado_pregunta CHECK (estado in ('ACTIVA', 'INACTIVA' , 'REVISION'))
);

CREATE TABLE alternativa (
	id_alternativa SERIAL PRIMARY KEY,
	id_pregunta INT NOT NULL,
	letra CHAR(1) NOT NULL,
	tipo VARCHAR(20) NOT NULL,

	contenido_texto TEXT,
	imagen_url TEXT,

	es_correcta BOOLEAN NOT NULL DEFAULT FALSE,
	fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

	CONSTRAINT fk_alternativa_pregunta
		FOREIGN KEY (id_pregunta) REFERENCES pregunta(id_pregunta) ON DELETE CASCADE,

	CONSTRAINT chk_letra 
		CHECK (letra IN ('A','B','C','D','E')),
	
	CONSTRAINT chk_tipo_alternativa
		CHECK (tipo IN ('TEXTO','IMAGEN')),

	CONSTRAINT uk_pregunta_letra UNIQUE (id_pregunta,letra),

	CONSTRAINT chk_contenido_alternativa CHECK (
	(tipo = 'TEXTO' AND contenido_texto IS NOT NULL AND imagen_url IS NULL)
	OR 
	(tipo = 'IMAGEN' AND imagen_url IS NOT NULL)
	)
);

CREATE TABLE examen (
    id_examen SERIAL PRIMARY KEY,
    id_categoria INT NOT NULL,

    codigo VARCHAR(30) NOT NULL UNIQUE,
    nombre VARCHAR(200) NOT NULL,

    total_preguntas INT NOT NULL DEFAULT 100,
    cantidad_versiones INT NOT NULL DEFAULT 1,

    aleatorizar_preguntas BOOLEAN NOT NULL DEFAULT TRUE,
    aleatorizar_alternativas BOOLEAN NOT NULL DEFAULT TRUE,

    estado VARCHAR(20) NOT NULL DEFAULT 'GENERADO',

    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_examen_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria_examen(id_categoria),

    CONSTRAINT chk_total_preguntas
        CHECK (total_preguntas = 100),

    CONSTRAINT chk_cantidad_versiones
        CHECK (cantidad_versiones >= 1),

    CONSTRAINT chk_estado_examen
        CHECK (estado IN ('GENERADO','ANULADO','EXPORTADO'))
);

CREATE TABLE examen_configuracion_curso (
    id_configuracion SERIAL PRIMARY KEY,
    id_examen INT NOT NULL,
    id_curso INT NOT NULL,

    cantidad_total INT NOT NULL,
    cantidad_facil INT,
    cantidad_medio INT,
    cantidad_dificil INT,

    CONSTRAINT fk_configuracion_examen
        FOREIGN KEY (id_examen)
        REFERENCES examen(id_examen)
        ON DELETE CASCADE,

    CONSTRAINT fk_configuracion_curso
        FOREIGN KEY (id_curso)
        REFERENCES curso(id_curso),

    CONSTRAINT chk_cantidad_total
        CHECK (cantidad_total > 0),

    CONSTRAINT chk_cantidades_dificultad
        CHECK (
            cantidad_facil IS NULL 
            OR cantidad_medio IS NULL 
            OR cantidad_dificil IS NULL
            OR cantidad_total = cantidad_facil + cantidad_medio + cantidad_dificil
        ),

    CONSTRAINT uk_examen_curso
        UNIQUE (id_examen, id_curso)
);

CREATE TABLE examen_version (
    id_version SERIAL PRIMARY KEY,
    id_examen INT NOT NULL,

    nombre_version VARCHAR(5) NOT NULL, -- A, B, C, D
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_version_examen
        FOREIGN KEY (id_examen)
        REFERENCES examen(id_examen)
        ON DELETE CASCADE,

    CONSTRAINT uk_examen_version
        UNIQUE (id_examen, nombre_version)
);

CREATE TABLE examen_pregunta (
    id_examen_pregunta SERIAL PRIMARY KEY,
    id_version INT NOT NULL,
    id_pregunta INT NOT NULL,

    numero_orden INT NOT NULL,
    alternativas_ordenadas JSONB,

    CONSTRAINT fk_examen_pregunta_version
        FOREIGN KEY (id_version)
        REFERENCES examen_version(id_version)
        ON DELETE CASCADE,

    CONSTRAINT fk_examen_pregunta_pregunta
        FOREIGN KEY (id_pregunta)
        REFERENCES pregunta(id_pregunta),

    CONSTRAINT uk_version_pregunta
        UNIQUE (id_version, id_pregunta),

    CONSTRAINT uk_version_orden
        UNIQUE (id_version, numero_orden)
);
CREATE TABLE historial_accion (
    id_historial SERIAL PRIMARY KEY,

    accion VARCHAR(50) NOT NULL,
    modulo VARCHAR(50) NOT NULL,
    descripcion TEXT,

    usuario VARCHAR(150),
    fecha TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    id_examen INT,
    id_pregunta INT,

    CONSTRAINT fk_historial_examen
        FOREIGN KEY (id_examen)
        REFERENCES examen(id_examen),

    CONSTRAINT fk_historial_pregunta
        FOREIGN KEY (id_pregunta)
        REFERENCES pregunta(id_pregunta)
);

CREATE TABLE categoria_curso_config (
    id_config SERIAL PRIMARY KEY,
    id_categoria INT NOT NULL,
    id_curso INT NOT NULL,
    cantidad_default INT,
    cantidad_facil_default INT,
    cantidad_medio_default INT,
    cantidad_dificil_default INT,
    estado VARCHAR(20) NOT NULL DEFAULT 'ACTIVO',

    CONSTRAINT fk_config_categoria
        FOREIGN KEY (id_categoria)
        REFERENCES categoria_examen(id_categoria),

    CONSTRAINT fk_config_curso
        FOREIGN KEY (id_curso)
        REFERENCES curso(id_curso),

    CONSTRAINT uk_categoria_curso_config
        UNIQUE (id_categoria, id_curso)
);

DROP TABLE IF EXISTS alternativa CASCADE;
DROP TABLE IF EXISTS pregunta CASCADE;
